% Boxed theorem-like environments: the heading sits on the top frame.
% All share one counter (numbered within the lecture); each has its own
% counter name so that \cref prints the right word.
\tcbset{thmbox/.style={enhanced,
  colback=#1!3!white,colframe=#1!75!black,boxrule=0.6pt,arc=1.2mm,
  left=2mm,right=2mm,top=3.2mm,bottom=1.6mm,
  before skip=11pt,after skip=9pt,
  fonttitle=\bfseries\small,coltitle=white,
  attach boxed title to top left={xshift=3mm,yshift=-2.4mm},
  boxed title style={colback=#1!75!black,colframe=#1!75!black,arc=0.8mm,boxrule=0pt,
    left=1.2mm,right=1.2mm,top=0.4mm,bottom=0.4mm}}}
\newcounter{thmctr}[chapter]
\renewcommand{\thethmctr}{\thechapter.\arabic{thmctr}}
\newcommand{\newboxthm}[3]{% #1 env name, #2 printed name, #3 colour
  \newcounter{#1}\expandafter\def\csname the#1\endcsname{\thethmctr}%
  \crefname{#1}{#2}{#2s}%
  \NewDocumentEnvironment{#1}{o}
    {\stepcounter{thmctr}\refstepcounter{#1}%
     \begin{tcolorbox}[thmbox=#3,
       title={#2~\thethmctr\IfValueT{##1}{\ \mdseries(##1)}}]}
    {\end{tcolorbox}}}
\newboxthm{definition}{Definition}{blue}

\newboxthm{lemma}{Lemma}{orange}
\newboxthm{corollary}{Corollary}{orange}
